// Google Sheets'ten gelen o uzun .csv linkini buraya yapıştır const googleSheetCSV = "https://docs.google.com/spreadsheets/d/1fWTOwORpoLjURDHAq9zBH4ivGZBlbGi4tZFuSh2QqVg/edit?usp=sharing"; async function updateDashboard() { try { // 'v=' parametresi tarayıcının önbelleğini (cache) atlatıp gerçek zamanlı veriyi çeker const res = await fetch(googleSheetCSV + "?v=" + Date.now()); const text = await res.text(); // CSV'yi JSON'a çeviren basit ve hızlı motor const rows = text.split('\n').slice(1); let totalCarbon = 0; let count = 0; markerLayer.clearLayers(); // Haritadaki eski ağaçları temizle rows.forEach(row => { const cols = row.split(','); // Sütunları ayır if (cols.length > 1) { // Sütun sıralarına göre (0: ID, 1: Tür, 2: Karbon, 3: Enlem, 4: Boylam gibi) const lat = parseFloat(cols[3]); const lng = parseFloat(cols[4]); const karbon = parseFloat(cols[2]) || 0; if (!isNaN(lat) && !isNaN(lng)) { totalCarbon += karbon; count++; // Haritaya ağaç işaretçisini koy L.marker([lat, lng], { icon: appleMarker }) .bindPopup(`${cols[1]}
Karbon: ${karbon} kg`) .addTo(markerLayer); } } }); // Paneldeki rakamları saniyeler içinde güncelle document.getElementById('total-co2').innerText = totalCarbon.toLocaleString('tr-TR'); document.getElementById('tree-count').innerText = count; } catch (e) { console.error("Veri senkronizasyon hatası..."); } } // 3 saniyede bir kontrol et (Sen Excel'e yazıp Enter'a bastığın an kiosk güncellenir) setInterval(updateDashboard, 3000); updateDashboard();
